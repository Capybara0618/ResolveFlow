def offline():
    os.environ.update(HF_HUB_OFFLINE='1', TRANSFORMERS_OFFLINE='1')
    import torch

    torch.set_num_threads(CONFIG['cpu_threads'])
    torch.set_num_interop_threads(1)


def tokenizer_audit(chunks, queries):
    from transformers import AutoTokenizer

    embedding = AutoTokenizer.from_pretrained(str(EMBEDDING), local_files_only=True)
    reranker = AutoTokenizer.from_pretrained(str(MODEL), local_files_only=True)
    chunk_max = max(len(embedding(c['text'], truncation=False)['input_ids']) for c in chunks)
    query_max = max(len(embedding(q['query'], truncation=False)['input_ids']) for q in queries)
    max_pair, pairs = 0, 0
    for query in queries:
        scoped = [c for c in chunks if c['subject'] == query['subject']]
        lengths = reranker([query['query']] * len(scoped), [c['text'] for c in scoped],
            truncation=False)['input_ids']
        max_pair = max(max_pair, max(map(len, lengths)))
        pairs += len(scoped)
    if max(chunk_max, query_max, max_pair) > 512:
        raise ValueError(f'fixed model window exceeded: {chunk_max}/{query_max}/{max_pair}')
    return {'embedding_chunk_max_tokens': chunk_max, 'embedding_query_max_tokens': query_max,
        'ce_pair_max_tokens': max_pair, 'ce_pairs_checked': pairs,
        'tokenization_only_no_ce_forward': True, 'silent_truncations': 0}


def prepare(output):
    if output.exists() or CACHE.exists():
        raise FileExistsError('preparation output or vector cache already exists; use check')
    offline()
    corpus, dataset = read(DATA / 'corpus.json'), read(DATA / 'queries.json')
    validated = validate_questions(dataset, corpus)
    chunks = build_chunks(corpus)
    window = tokenizer_audit(chunks, dataset['queries'])
    started = time.perf_counter()
    encoder = embedder(str(EMBEDDING))
    vectors = encoder.embed([c['text'] for c in chunks])
    build_ms = (time.perf_counter() - started) * 1000
    if encoder._encoder().max_seq_length < window['embedding_chunk_max_tokens']:
        raise ValueError('embedding runtime window smaller than audited text')
    indexed = {'version': 'customer-rag-quantification-v3', 'config': CONFIG,
        'corpus_sha256': sha((DATA / 'corpus.json').read_bytes()), 'chunks': chunks}
    if (DATA / 'chunks.json').exists():
        if read(DATA / 'chunks.json') != indexed:
            raise ValueError('tracked chunk index differs from deterministic source build')
    else:
        write_new(DATA / 'chunks.json', indexed)
    write_new(CACHE, {'chunks_sha256': sha((DATA / 'chunks.json').read_bytes()),
        'model': encoder.model, 'revision': encoder.revision,
        'vectors': {c['id']: v for c, v in zip(chunks, vectors, strict=True)}})
    by_subject = {s: {'documents': len({c['source_id'] for c in chunks if c['subject'] == s}),
        'chunks': sum(c['subject'] == s for c in chunks)} for s in sorted({c['subject'] for c in chunks})}
    result = {'stage': 'PREPARED_NOT_EVALUATED', **validated, 'chunks': len(chunks),
        'by_subject': by_subject, 'window_audit': window, 'document_encoding_ms': build_ms,
        'embedding_dimension': encoder.dimension, 'embedding_runtime_window': encoder._encoder().max_seq_length,
        'vectors_sha256': sha(CACHE.read_bytes()), 'chunks_sha256': sha((DATA / 'chunks.json').read_bytes()),
        'new_query_rankings': 0, 'new_ce_forwards': 0, 'new_generation_api_calls': 0}
    write_new(output, result)
    return result


def frozen_assets():
    paths = ['docs/customer-rag-quantification-v3.md', 'scripts/customer_rag_dataset_v3.py',
        'scripts/customer_rag_questions_v3.py', 'scripts/customer_rag_benchmark_v3.py',
        'scripts/evaluation/customer_policy_benchmark_v3.py',
        'scripts/evaluation/customer_rerank.py', 'scripts/evaluation/customer_rerank_gpu.py',
        'scripts/evaluation/customer_rerank_gpu_worker.py', 'scripts/customer_quantification_rag.py',
        'scripts/customer_rag_benchmark.py', 'scripts/customer_rag_rerank.py', 'scripts/policy_documents.py',
        'agent/src/resolveflow/policies/retrieval.py', 'agent/src/resolveflow/policies/index.py',
        'agent/src/resolveflow/harness/token_counting.py', 'agent/uv.lock',
        'agent/src/resolveflow/assets/deepseek-v41/tokenizer.json.gz',
        'agent/tests/unit/test_customer_rag_dataset_v3.py', 'agent/tests/unit/test_customer_rag_questions_v3.py',
        'agent/tests/unit/test_customer_policy_benchmark_v3.py', 'agent/tests/unit/test_customer_rag_benchmark_v3.py']
    paths += [p.relative_to(ROOT).as_posix() for p in sorted(DATA.glob('*.json')) if p != PLAN]
    paths += [CACHE.relative_to(ROOT).as_posix()]
    for source in read(DATA / 'sources.json'):
        paths.extend([source['snapshot_path'], source['capture_path']])
    assets = {'files': {p: sha((ROOT / p).read_bytes()) for p in paths}}
    for name, directory in [('embedding_files', EMBEDDING), ('reranker_files', MODEL)]:
        assets[name] = {p.relative_to(directory).as_posix(): sha(p.read_bytes())
            for p in sorted(directory.rglob('*')) if p.is_file() and not p.name.endswith('.lock')}
        if not assets[name]:
            raise ValueError('local pinned model assets absent')
    return assets


def freeze():
    dataset = read(DATA / 'queries.json')
    validate_questions(dataset, read(DATA / 'corpus.json'))
    if read(DATA / 'chunks.json')['chunks'] != build_chunks(read(DATA / 'corpus.json')):
        raise ValueError('deterministic index differs')
    write_new(PLAN, {'version': 'customer-rag-quantification-v3', 'config': CONFIG,
        'variants': VARIANTS, 'primary_baseline': 'dense', 'primary_candidate': 'hybrid-rrf-ce',
        'strong_control': 'bm25', 'query_split': {'development': 40, 'acceptance': 60},
        'quality_denominators': {'development': 32, 'acceptance': 48},
        'independent_human_holdout': False, 'production_adoption': False,
        'answer_generation_enabled': False, **frozen_assets()})


def check():
    plan = read(PLAN)
    if plan['config'] != CONFIG or plan['variants'] != VARIANTS or plan['answer_generation_enabled'] is not False:
        raise ValueError('frozen protocol differs')
    for key, directory in [('files', ROOT), ('embedding_files', EMBEDDING), ('reranker_files', MODEL)]:
        for name, expected in plan[key].items():
            if sha((directory / name).read_bytes()) != expected:
                raise ValueError('frozen asset changed: ' + name)
    return plan


def summarize(rows, queries, rounds=3):
    expected = {(q['id'], v, r) for q in queries for v in VARIANTS for r in range(1, rounds + 1)}
    actual = [(r['id'], r['variant'], r['repeat']) for r in rows]
    if set(actual) != expected or len(set(actual)) != len(actual):
        raise ValueError('incomplete/duplicate retrieval matrix')
    if any(not math.isfinite(r['elapsed_ms']) or r['elapsed_ms'] < 0 for r in rows):
        raise ValueError('invalid latency')
    answerable = {q['id'] for q in queries if q['answerability'] == 'ANSWERABLE'}
    result = {}
    for variant in VARIANTS:
        selected = [r for r in rows if r['variant'] == variant]
        scored = [r for r in selected if r['repeat'] == 1 and r['id'] in answerable]
        values = sorted(r['elapsed_ms'] for r in selected)
        complete = sum(r['quality']['3']['complete'] for r in scored)
        result[variant] = {'answerable_questions': len(scored), 'complete_at_3_count': complete,
            'complete_at_3_rate': complete / len(scored) if scored else None,
            'top1_any_count': sum(r['quality']['1']['any_required'] for r in scored),
            'complete_at_1_count': sum(r['quality']['1']['complete'] for r in scored),
            'complete_at_5_count': sum(r['quality']['5']['complete'] for r in scored),
            'timed_requests': len(values), 'mean_ms': mean(values),
            'p50_ms': values[math.ceil(len(values) * .5) - 1],
            'p95_ms': values[math.ceil(len(values) * .95) - 1],
            'mean_body_tokens': mean(r['packet']['body_tokens'] for r in selected),
            'stable_packets': all(len({tuple(r['packet']['selected_chunks']) for r in selected if r['id'] == q['id']}) == 1
                and len({r['packet']['body_sha256'] for r in selected if r['id'] == q['id']}) == 1 for q in queries),
            'by_category': {category: {'questions': sum(r['category'] == category for r in scored),
                'complete': sum(r['quality']['3']['complete'] for r in scored if r['category'] == category)}
                for category in sorted({q['category'] for q in queries if q['answerability'] == 'ANSWERABLE'})}}
    return result


def run(output, split, *, smoke=False):
    plan = check()
    if output.exists() or output.with_suffix('.jsonl').exists():
        raise FileExistsError(output)
    offline()
    chunks = read(DATA / 'chunks.json')['chunks']
    vectors = read(CACHE)['vectors']
    queries = [q for q in read(DATA / 'queries.json')['queries'] if q['split'] == split]
    if smoke:
        original = queries[0]
        queries = [{**public_query(original), 'id': 'GENERIC_SMOKE', 'category': 'generic',
            'answerability': 'NOT_IN_CORPUS', 'required_evidence': [],
            'query': '退货前需要准备哪些材料？'}]
    started = time.perf_counter()
    encoder, tokenizer = embedder(str(EMBEDDING)), load_tokenizer()
    worker = GPUWorker(GPU_PYTHON, MODEL, output.with_suffix('.gpu.log'))
    try:
        scorer = LocalReranker(worker, synchronize=worker.synchronize)
        encoder.embed(['售后政策适用范围'])
        # Warmup uses generic text, never a new labeled query.
        from resolveflow.policies.index import PolicyHit
        warm = PolicyHit('warm', 'M', 'v', 'h', 'r', 'c', 'h', 'title', '人工核查后处理。', 0., 'dense')
        scorer.rank('售后核查', [warm])
        startup_ms = (time.perf_counter() - started) * 1000
        rows = []
        rounds = 1 if smoke else plan['config']['rounds']
        with output.with_suffix('.jsonl').open('x', encoding='utf-8', newline='\n') as journal:
            journal.write(json.dumps({'manifest_sha256': sha(PLAN.read_bytes()), 'split': split,
                'generic_smoke_only': smoke, 'rounds': rounds}) + '\n')
            for repeat in range(1, rounds + 1):
                order = VARIANTS[repeat - 1:] + VARIANTS[:repeat - 1]
                for query in queries:
                    for variant in order:
                        tick = time.perf_counter()
                        ranked, timing = retrieve(public_query(query), chunks, vectors, encoder, scorer, variant)
                        packet = assemble(ranked, tokenizer)
                        elapsed = (time.perf_counter() - tick) * 1000
                        packet1, packet5 = assemble(ranked, tokenizer, k=1), assemble(ranked, tokenizer, k=5)
                        row = {'id': query['id'], 'category': query['category'], 'variant': variant,
                            'repeat': repeat, 'elapsed_ms': elapsed, 'timing': timing,
                            'hits': [{'id': c['id'], 'score': float(s)} for c, s in ranked],
                            'packet': packet, 'quality': {str(k): quality(query, p)
                                for k, p in [(1, packet1), (3, packet), (5, packet5)]}}
                        rows.append(row)
                        journal.write(json.dumps(row, ensure_ascii=False) + '\n')
                        journal.flush()
                print(json.dumps({'completed_round': repeat, 'requests': len(rows)}), flush=True)
        check()
        result = {'version': plan['version'], 'manifest_sha256': sha(PLAN.read_bytes()),
            'split': split, 'generic_smoke_only': smoke, 'startup_ms': startup_ms,
            'gpu': worker.ready, 'new_generation_api_calls': 0,
            'latency_scope': 'Query to Top3 evidence; startup/index/HTTP/PG/answer generation/scoring excluded',
            'summary': summarize(rows, queries, rounds), 'rows': rows}
        write_new(output, result)
        return {'requests': len(rows), 'generic_smoke_only': smoke, 'new_generation_api_calls': 0}
    finally:
        worker.close()


if __name__ == '__main__':
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('mode', choices=['prepare', 'freeze', 'check', 'smoke', 'execute'])
    parser.add_argument('--split', choices=['development', 'acceptance'])
    parser.add_argument('--output', type=Path)
    args = parser.parse_args()
    if args.mode == 'prepare' and args.output:
        print(json.dumps(prepare(args.output)))
    elif args.mode == 'freeze':
        freeze()
        print(json.dumps({'frozen': True, 'new_generation_api_calls': 0}))
    elif args.mode == 'check':
        print(json.dumps({'version': check()['version'], 'new_generation_api_calls': 0}))
    elif args.mode in {'smoke', 'execute'} and args.output and (args.mode == 'smoke' or args.split):
        print(json.dumps(run(args.output, args.split or 'development', smoke=args.mode == 'smoke')))
    else:
        parser.error('prepare/smoke/execute require a new --output; execute also requires explicit --split')
